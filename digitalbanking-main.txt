package tutorjava.digitalbanking.account;

import jakarta.persistence.Column;
import jakarta.persistence.Entity;
import jakarta.persistence.EnumType;
import jakarta.persistence.Enumerated;
import jakarta.persistence.Table;
import lombok.AccessLevel;
import lombok.Getter;
import lombok.NoArgsConstructor;
import tutorjava.digitalbanking.common.entity.AuditableEntity;
import tutorjava.digitalbanking.common.exception.BusinessException;
import tutorjava.digitalbanking.common.exception.ErrorCode;

import java.math.BigDecimal;
import java.util.UUID;

/**
 * Rich domain object: balance can only change through credit()/debit(), which
 * enforce the
 * business invariants (active account, no overdraft). The DB CHECK (balance >=
 * 0) is the safety net.
 */
@Getter
@Entity
@Table(name = "accounts")
@NoArgsConstructor(access = AccessLevel.PROTECTED)
public class Account extends AuditableEntity {

    @Column(name = "account_number", nullable = false, updatable = false, length = 10)
    private String accountNumber;

    @Column(name = "user_id", nullable = false, updatable = false)
    private UUID userId;

    @Column(name = "account_name", nullable = false, length = 50)
    private String accountName;

    @Column(name = "currency", nullable = false, updatable = false, length = 3)
    private String currency;

    @Column(name = "balance", nullable = false, precision = 19, scale = 2)
    private BigDecimal balance;

    @Enumerated(EnumType.STRING)
    @Column(name = "status", nullable = false, length = 20)
    private AccountStatus status;

    public static Account open(UUID userId, String accountNumber, String accountName, String currency) {
        Account account = new Account();
        account.userId = userId;
        account.accountNumber = accountNumber;
        account.accountName = accountName;
        account.currency = currency;
        account.balance = BigDecimal.ZERO.setScale(2);
        account.status = AccountStatus.ACTIVE;
        return account;
    }

    public boolean isOwnedBy(UUID candidateUserId) {
        return userId.equals(candidateUserId);
    }

    public void credit(BigDecimal amount) {
        requireActive();
        balance = balance.add(amount);
    }

    public void debit(BigDecimal amount) {
        requireActive();
        if (balance.compareTo(amount) < 0) {
            throw new BusinessException(ErrorCode.INSUFFICIENT_BALANCE);
        }
        balance = balance.subtract(amount);
    }

    private void requireActive() {
        if (status != AccountStatus.ACTIVE) {
            throw new BusinessException(ErrorCode.ACCOUNT_NOT_ACTIVE);
        }
    }
}

package tutorjava.digitalbanking.account;

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;
import tutorjava.digitalbanking.common.exception.BusinessException;
import tutorjava.digitalbanking.common.exception.ErrorCode;

import java.util.HashSet;
import java.util.LinkedHashMap;
import java.util.Map;
import java.util.Set;
import java.util.UUID;

/**
 * Acquires row locks on accounts. Propagation.MANDATORY guarantees this is only
 * ever called
 * from inside a transaction (a lock outside a transaction would be released
 * immediately).
 */
@Service
@RequiredArgsConstructor
public class AccountLockService {

    private final AccountRepository accountRepository;

    @Transactional(propagation = Propagation.MANDATORY)
    public Map<String, Account> lock(Set<String> accountNumbers) {
        Map<String, Account> locked = new LinkedHashMap<>();
        accountRepository.findAllForUpdate(accountNumbers).forEach(a -> locked.put(a.getAccountNumber(), a));
        if (!locked.keySet().containsAll(new HashSet<>(accountNumbers))) {
            throw new BusinessException(ErrorCode.ACCOUNT_NOT_FOUND);
        }
        return locked;
    }

    /**
     * Locks one account and verifies it belongs to the user (foreign accounts look
     * like "not found").
     */
    @Transactional(propagation = Propagation.MANDATORY)
    public Account lockOwned(String accountNumber, UUID userId) {
        Account account = lock(Set.of(accountNumber)).get(accountNumber);
        if (!account.isOwnedBy(userId)) {
            throw new BusinessException(ErrorCode.ACCOUNT_NOT_FOUND);
        }
        return account;
    }
}

package tutorjava.digitalbanking.account;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import tutorjava.digitalbanking.account.dto.AccountResponse;
import tutorjava.digitalbanking.account.dto.CreateAccountRequest;
import tutorjava.digitalbanking.common.exception.BusinessException;
import tutorjava.digitalbanking.common.exception.ErrorCode;
import tutorjava.digitalbanking.config.BankingProperties;

import java.util.List;
import java.util.UUID;

@Slf4j
@Service
@RequiredArgsConstructor
public class AccountService {

    private static final int MAX_NUMBER_ATTEMPTS = 5;

    private final AccountRepository accountRepository;
    private final AccountNumberGenerator numberGenerator;
    private final BankingProperties bankingProperties;

    @Transactional
    public AccountResponse open(UUID userId, CreateAccountRequest request) {
        if (accountRepository.countByUserId(userId) >= bankingProperties.maxAccountsPerUser()) {
            throw new BusinessException(ErrorCode.ACCOUNT_LIMIT_REACHED);
        }
        Account account = Account.open(userId, uniqueAccountNumber(), request.accountName().trim(),
                bankingProperties.currency());
        accountRepository.save(account);
        log.atInfo()
                .addKeyValue("userId", userId)
                .addKeyValue("accountNumber", account.getAccountNumber())
                .log("Account opened");
        return AccountResponse.from(account);
    }

    @Transactional(readOnly = true)
    public List<AccountResponse> listOwn(UUID userId) {
        return accountRepository.findByUserIdOrderByCreatedAtDesc(userId).stream()
                .map(AccountResponse::from)
                .toList();
    }

    @Transactional(readOnly = true)
    public AccountResponse getOwn(UUID userId, String accountNumber) {
        return accountRepository.findByAccountNumberAndUserId(accountNumber, userId)
                .map(AccountResponse::from)
                .orElseThrow(() -> new BusinessException(ErrorCode.ACCOUNT_NOT_FOUND));
    }

    private String uniqueAccountNumber() {
        for (int attempt = 0; attempt < MAX_NUMBER_ATTEMPTS; attempt++) {
            String candidate = numberGenerator.next();
            if (!accountRepository.existsByAccountNumber(candidate)) {
                return candidate; // the UNIQUE constraint still guards against a race
            }
        }
        throw new IllegalStateException("Could not generate a unique account number");
    }
}

        + "Mobile (header X-Client-Type: mobile): refresh token in the JSON body.")
public class AuthController {

    private static final String CLIENT_TYPE_HEADER = "X-Client-Type";
    private static final String CLIENT_TYPE_MOBILE = "mobile";

    private final AuthService authService;
    private final RefreshTokenCookies cookies;

    @PostMapping("/register")
    @ResponseStatus(HttpStatus.CREATED)
    @Operation(summary = "Register a new customer")
    @ApiResponse(responseCode = "409", description = "Email already registered", content = @Content(schema = @Schema(implementation = ApiError.class)))
    public UserResponse register(@Valid @RequestBody RegisterRequest request) {
        return authService.register(request);
    }

    @Operation(summary = "Login. Returns a short-lived access token and starts a refresh-token session", description = "Web (default): the refresh token is set as an HttpOnly cookie and omitted from the body. "
            + "Mobile: send `X-Client-Type: mobile` to receive `refreshToken` in the body.")
    @ApiResponse(responseCode = "401", description = "Invalid credentials", content = @Content(schema = @Schema(implementation = ApiError.class)))
    @ApiResponse(responseCode = "423", description = "User temporarily locked after repeated failures", content = @Content(schema = @Schema(implementation = ApiError.class)))
    public ResponseEntity<AuthResponse> login(
            @Valid @RequestBody LoginRequest request,
            @Parameter(description = "Use 'mobile' to get the refresh token in the response body") @RequestHeader(name = CLIENT_TYPE_HEADER, required = false) String clientType) {
        boolean bodyDelivery = CLIENT_TYPE_MOBILE.equalsIgnoreCase(clientType);
        return tokenResponse(authService.login(request), bodyDelivery);
    }

    @PostMapping("/refresh")
    @Operation(summary = "Get a new access token (the refresh token is rotated)", description = "Web: call with no body; the browser sends the HttpOnly cookie. "
            + "Mobile: send `{\"refreshToken\": \"...\"}`. Every refresh token is single-use: "
            + "replaying an already used token revokes the whole session.")
    @ApiResponse(responseCode = "401", description = "Refresh token invalid, expired or already used", content = @Content(schema = @Schema(implementation = ApiError.class)))
    @ApiResponse(responseCode = "403", description = "Cookie request from an untrusted Origin", content = @Content(schema = @Schema(implementation = ApiError.class)))
    public ResponseEntity<AuthResponse> refresh(@Valid @RequestBody(required = false) RefreshRequest body,
            HttpServletRequest http) {
        String fromBody = bodyToken(body);
        boolean bodyDelivery = fromBody != null;
        String rawToken;
        if (bodyDelivery) {
            rawToken = fromBody;
        } else {
            cookies.requireTrustedOrigin(http); // cookie => CSRF defence
            rawToken = cookies.read(http)
                    .orElseThrow(() -> new BusinessException(ErrorCode.INVALID_REFRESH_TOKEN));
        }
        return tokenResponse(authService.refresh(rawToken), bodyDelivery);
    }

    @PostMapping("/logout")
    @Operation(summary = "Logout: revoke the current session (idempotent)", description = "Web: no body, the cookie identifies the session and is cleared. "
            + "Mobile: send `{\"refreshToken\": \"...\"}`. Always returns 204. "
            + "The access token already issued stays valid until it expires (max 15 minutes).")
    @ApiResponse(responseCode = "204", description = "Logged out")
    public ResponseEntity<Void> logout(@Valid @RequestBody(required = false) RefreshRequest body,
            HttpServletRequest http) {
        String fromBody = bodyToken(body);
        if (fromBody != null) {
            authService.logout(fromBody);
            return ResponseEntity.noContent().build();
        }
        cookies.requireTrustedOrigin(http);
        cookies.read(http).ifPresent(authService::logout);
        return ResponseEntity.noContent().header(HttpHeaders.SET_COOKIE, cookies.clear().toString()).build();
    }

    @PostMapping("/logout-all")
    @SecurityRequirement(name = OpenApiConfig.BEARER_SCHEME)

    @ApiResponse(responseCode = "204", description = "All sessions revoked")

            @Parameter(hidden = true) @AuthenticationPrincipal AuthenticatedUser principal) {
        authService.logoutEverywhere(principal.id());
        return ResponseEntity.noContent().header(HttpHeaders.SET_COOKIE, cookies.clear().toString()).build();
    }

    // ------------------------------------------------------------------ helpers
    private ResponseEntity<AuthResponse> tokenResponse(AuthResult result, boolean bodyDelivery) {
        if (bodyDelivery) {
            return ResponseEntity.ok(AuthResponse.from(result.accessToken(), result.refreshToken().value()));
        }
        ResponseCookie cookie = cookies.create(result.refreshToken());
        return ResponseEntity.ok()
                .header(HttpHeaders.SET_COOKIE, cookie.toString())
                .body(AuthResponse.from(result.accessToken(), null));
    }

    private String bodyToken(RefreshRequest body) {
        return body != null && body.refreshToken() != null && !body.refreshToken().isBlank()
                ? body.refreshToken().trim()
                : null;
    }
}

package tutorjava.digitalbanking.auth;

import lombok.extern.slf4j.Slf4j;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import tutorjava.digitalbanking.auth.dto.LoginRequest;
import tutorjava.digitalbanking.auth.dto.RegisterRequest;
import tutorjava.digitalbanking.common.exception.BusinessException;
import tutorjava.digitalbanking.common.exception.ErrorCode;
import tutorjava.digitalbanking.config.AuthProperties;
import tutorjava.digitalbanking.security.JwtService;
import tutorjava.digitalbanking.user.User;
import tutorjava.digitalbanking.user.UserRepository;
import tutorjava.digitalbanking.user.dto.UserResponse;

import java.time.Instant;
import java.util.Locale;
import java.util.UUID;

@Slf4j
@Service
public class AuthService {

    private final UserRepository userRepository;
    private final PasswordEncoder passwordEncoder;
    private final JwtService jwtService;
    private final RefreshTokenService refreshTokenService;
    private final AuthProperties authProperties;

    /**
     * Used to spend the same CPU time when the email is unknown (mitigates user
     * enumeration by timing).
     */
    private final String dummyHash;

    public AuthService(UserRepository userRepository, PasswordEncoder passwordEncoder,
            JwtService jwtService, RefreshTokenService refreshTokenService,
            AuthProperties authProperties) {
        this.userRepository = userRepository;
        this.passwordEncoder = passwordEncoder;
        this.jwtService = jwtService;
        this.refreshTokenService = refreshTokenService;
        this.authProperties = authProperties;
        this.dummyHash = passwordEncoder.encode("timing-attack-mitigation-dummy");
    }

    @Transactional
    public UserResponse register(RegisterRequest request) {
        String email = normalizeEmail(request.email());
        if (userRepository.existsByEmail(email)) {
            throw new BusinessException(ErrorCode.EMAIL_ALREADY_REGISTERED);
        }
        User user = User.register(email, request.fullName().trim(), passwordEncoder.encode(request.password()));
        try {
            // flush now so a concurrent duplicate surfaces here as a clean business error
            userRepository.saveAndFlush(user);
        } catch (DataIntegrityViolationException e) {
            throw new BusinessException(ErrorCode.EMAIL_ALREADY_REGISTERED);
        }
        log.atInfo().addKeyValue("userId", user.getId()).log("User registered");
        return UserResponse.from(user);
    }

    /**
     * Intentionally not @Transactional: the failed-attempt counter must be
     * committed
     * even though the method ends by throwing an exception.
     */
    public AuthResult login(LoginRequest request) {
        String email = normalizeEmail(request.email());
        User user = userRepository.findByEmail(email).orElse(null);

        if (user == null) {
            passwordEncoder.matches(request.password(), dummyHash);
            log.atWarn().log("Login failed: unknown email");
            throw new BusinessException(ErrorCode.INVALID_CREDENTIALS);
        }

        Instant now = Instant.now();
        if (user.isLocked(now)) {
            log.atWarn().addKeyValue("userId", user.getId()).log("Login rejected: user temporarily locked");
            throw new BusinessException(ErrorCode.USER_LOCKED);
        }

        boolean passwordOk = passwordEncoder.matches(request.password(), user.getPasswordHash());
        if (!passwordOk || !user.isEnabled()) {
            user.registerFailedLogin(authProperties.maxFailedLoginAttempts(), authProperties.lockDuration(), now);
            userRepository.save(user);
            log.atWarn().addKeyValue("userId", user.getId()).log("Login failed: bad credentials");
            throw new BusinessException(ErrorCode.INVALID_CREDENTIALS);
        }

        if (user.hasLoginFailureState()) {
            user.resetLoginFailures();
            user = userRepository.save(user);
        }
        log.atInfo().addKeyValue("userId", user.getId()).log("Login succeeded");
        return new AuthResult(jwtService.generateAccessToken(user), refreshTokenService.startSession(user.getId()));
    }

    /**
     * Exchanges a valid refresh token for a new access token + a rotated refresh
     * token.
     */
    public AuthResult refresh(String rawRefreshToken) {
        RefreshTokenService.RotatedSession session = refreshTokenService.rotate(rawRefreshToken);
        return new AuthResult(jwtService.generateAccessToken(session.user()), session.refreshToken());
    }

    /**
     * Ends the session the token belongs to. Never fails for unknown tokens
     * (idempotent).
     */
    public void logout(String rawRefreshToken) {
        refreshTokenService.revokeSession(rawRefreshToken);
    }

    public void logoutEverywhere(UUID userId) {
        refreshTokenService.revokeAllForUser(userId);
    }

    private String normalizeEmail(String email) {
        return email.trim().toLowerCase(Locale.ROOT);
    }
}

import org.springframework.stereotype.Component;
import tutorjava.digitalbanking.common.exception.BusinessException;

import tutorjava.digitalbanking.config.RefreshTokenProperties;

import java.time.Duration;
import java.time.Instant;
import java.util.Arrays;
import java.util.Optional;

/**
 * Browser delivery of the refresh token: HttpOnly (invisible to JavaScript, so
 * XSS cannot steal it),
 * Secure, SameSite, and scoped to the auth path so it is not sent with ordinary
 * API calls.
 * Because cookies are sent automatically, cookie-based endpoints also verify
 * the Origin header (CSRF defence).
 */
@Component
@RequiredArgsConstructor
public class RefreshTokenCookies {

    private final RefreshTokenProperties properties;
    private final CorsProperties corsProperties;

    public Optional<String> read(HttpServletRequest request) {
        Cookie[] cookies = request.getCookies();
        if (cookies == null) {
            return Optional.empty();
        }
        return Arrays.stream(cookies)
                .filter(c -> properties.cookieName().equals(c.getName()))
                .map(Cookie::getValue)
                .filter(v -> v != null && !v.isBlank())
                .findFirst();
    }

    public ResponseCookie create(IssuedRefreshToken token) {
        Duration maxAge = Duration.between(Instant.now(), token.expiresAt());
        return base(token.value()).maxAge(maxAge.isNegative() ? Duration.ZERO : maxAge).build();
    }

    public ResponseCookie clear() {
        return base("").maxAge(Duration.ZERO).build();
    }

    /**
     * Browsers always send Origin on cross-origin POSTs. If it is present it must
     * be one of our allowed
     * origins. Requests without Origin come from non-browser clients, which are not
     * exposed to CSRF.
     */
    public void requireTrustedOrigin(HttpServletRequest request) {
        String origin = request.getHeader("Origin");
        if (origin != null && !corsProperties.allowedOrigins().contains(origin)) {
            throw new BusinessException(ErrorCode.FORBIDDEN);
        }
    }

    private ResponseCookie.ResponseCookieBuilder base(String value) {
        return ResponseCookie.from(properties.cookieName(), value)
                .httpOnly(true)
                .secure(properties.cookieSecure())
                .sameSite(properties.cookieSameSite())
                .path(properties.cookiePath());
    }
}

package tutorjava.digitalbanking.auth;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import tutorjava.digitalbanking.common.exception.BusinessException;
import tutorjava.digitalbanking.common.exception.ErrorCode;
import tutorjava.digitalbanking.common.util.Sha256;
import tutorjava.digitalbanking.config.RefreshTokenProperties;
import tutorjava.digitalbanking.user.User;
import tutorjava.digitalbanking.user.UserRepository;

import java.security.SecureRandom;
import java.time.Instant;
import java.util.Base64;
import java.util.UUID;

/**
 * Refresh-token lifecycle with rotation and reuse detection.
 *
 * - login -> new family (a "session") with its first token
 * - refresh -> the presented token is atomically revoked and replaced by a new
 * one in the same family
 * - reuse -> presenting a token that was already used/revoked means it was
 * probably stolen:
 * the whole family is revoked, so the thief AND the victim must log in again
 * - logout -> the whole family is revoked
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class RefreshTokenService {

    private static final int TOKEN_BYTES = 32; // 256 bits of entropy

    private final RefreshTokenRepository repository;
    private final UserRepository userRepository;
    private final RefreshTokenProperties properties;
    private final SecureRandom secureRandom = new SecureRandom();

    public record RotatedSession(User user, IssuedRefreshToken refreshToken) {
    }

    @Transactional
    public IssuedRefreshToken startSession(UUID userId) {
        Instant now = Instant.now();
        return issue(userId, UUID.randomUUID(), now, now);
    }

    /**
     * noRollbackFor: when reuse is detected we revoke the family and THEN throw
     * 401. The revocation
     * must be committed, so this exception must not roll the transaction back.
     */
    @Transactional(noRollbackFor = BusinessException.class)
    public RotatedSession rotate(String rawToken) {
        Instant now = Instant.now();
        RefreshToken stored = repository.findByTokenHash(Sha256.hex(rawToken))
                .orElseThrow(() -> new BusinessException(ErrorCode.INVALID_REFRESH_TOKEN));

        if (stored.getRevokedAt() != null) {
            throw reuseDetected(stored, now);
        }
        if (stored.isExpired(now)) {
            throw new BusinessException(ErrorCode.INVALID_REFRESH_TOKEN);
        }
        // Atomic claim: if two requests present the same token concurrently only one
        // gets 1 here.
        if (repository.revokeIfActive(stored.getId(), now) == 0) {
            throw reuseDetected(stored, now);
        }

        User user = userRepository.findById(stored.getUserId()).filter(User::isEnabled).orElse(null);
        if (user == null) {
            repository.revokeFamily(stored.getFamilyId(), now);
            throw new BusinessException(ErrorCode.INVALID_REFRESH_TOKEN);
        }

        IssuedRefreshToken next = issue(user.getId(), stored.getFamilyId(), stored.getSessionStartedAt(), now);
        log.atInfo().addKeyValue("userId", user.getId()).log("Refresh token rotated");
        return new RotatedSession(user, next);
    }

    /** Idempotent: unknown/expired/already revoked tokens are silently ignored. */
    @Transactional
    public void revokeSession(String rawToken) {
        repository.findByTokenHash(Sha256.hex(rawToken)).ifPresent(token -> {
            repository.revokeFamily(token.getFamilyId(), Instant.now());
            log.atInfo().addKeyValue("userId", token.getUserId()).log("Session revoked (logout)");
        });
    }

    @Transactional
    public void revokeAllForUser(UUID userId) {
        int revoked = repository.revokeAllForUser(userId, Instant.now());
        log.atInfo().addKeyValue("userId", userId).addKeyValue("revokedTokens", revoked)
                .log("All sessions revoked (logout everywhere)");
    }

    private IssuedRefreshToken issue(UUID userId, UUID familyId, Instant sessionStartedAt, Instant now) {
        Instant slidingExpiry = now.plus(properties.ttl());
        Instant absoluteExpiry = sessionStartedAt.plus(properties.absoluteTtl());
        Instant expiresAt = slidingExpiry.isBefore(absoluteExpiry) ? slidingExpiry : absoluteExpiry;

        String raw = generateRawToken();
        repository.save(RefreshToken.create(userId, familyId, Sha256.hex(raw), sessionStartedAt, expiresAt));
        return new IssuedRefreshToken(raw, expiresAt);
    }

    private BusinessException reuseDetected(RefreshToken stored, Instant now) {
        repository.revokeFamily(stored.getFamilyId(), now);
        log.atWarn()
                .addKeyValue("userId", stored.getUserId())
                .addKeyValue("familyId", stored.getFamilyId())
                .log("Refresh token reuse detected: session family revoked");
        return new BusinessException(ErrorCode.INVALID_REFRESH_TOKEN);
    }

    private String generateRawToken() {
        byte[] bytes = new byte[TOKEN_BYTES];
        secureRandom.nextBytes(bytes);
        return Base64.getUrlEncoder().withoutPadding().encodeToString(bytes);
    }
}

import jakarta.servlet.http.HttpServletRequest;
import lombok.extern.slf4j.Slf4j;
import org.springframework.dao.ConcurrencyFailureException;
import org.springframework.dao.DataIntegrityViolationException;

import org.springframework.http.HttpStatusCode;
import org.springframework.http.ProblemDetail;
import org.springframework.http.ResponseEntity;
import org.springframework.security.access.AccessDeniedException;
import org.springframework.security.core.AuthenticationException;
import org.springframework.web.ErrorResponse;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.context.request.ServletWebRequest;
import org.springframework.web.context.request.WebRequest;
import org.springframework.web.servlet.mvc.method.annotation.ResponseEntityExceptionHandler;
import tutorjava.digitalbanking.common.response.ApiError;

import java.util.List;

/**
 * Central error translation. Extends Spring's ResponseEntityExceptionHandler so
 * framework
 * exceptions (405, 415, malformed JSON, missing header, ...) also use the same
 * ApiError shape.
 * Internal details are logged, never returned to the client.
 */
@Slf4j
@RestControllerAdvice
public class GlobalExceptionHandler extends ResponseEntityExceptionHandler {

    // ---------------------------------------------------------------- domain
    // errors
    @ExceptionHandler(BusinessException.class)
    public ResponseEntity<ApiError> handleBusiness(BusinessException ex, HttpServletRequest request) {
        ErrorCode code = ex.getErrorCode();
        log.atWarn()
                .addKeyValue("errorCode", code.name())
                .addKeyValue("path", request.getRequestURI())
                .log("Business rule rejected the request: {}", ex.getMessage());
        return build(code, ex.getMessage(), request.getRequestURI());
    }

    // ---------------------------------------------------------------- persistence
    // errors
    /**
     * Optimistic-lock failures, deadlocks, lock timeouts: safe for the client to
     * retry.
     */
    @ExceptionHandler(ConcurrencyFailureException.class)
    public ResponseEntity<ApiError> handleConcurrency(ConcurrencyFailureException ex, HttpServletRequest request) {
        log.atWarn().addKeyValue("path", request.getRequestURI())
                .log("Concurrent update detected: {}", ex.getClass().getSimpleName());
        return build(ErrorCode.CONCURRENT_UPDATE, ErrorCode.CONCURRENT_UPDATE.getDefaultMessage(),
                request.getRequestURI());
    }

    /**
     * Unique/FK/CHECK violation that slipped past service-level checks. Never
     * expose SQL details.
     */
    @ExceptionHandler(DataIntegrityViolationException.class)
    public ResponseEntity<ApiError> handleDataIntegrity(DataIntegrityViolationException ex,
            HttpServletRequest request) {
        log.atWarn().addKeyValue("path", request.getRequestURI())
                .log("Data integrity violation", ex);
        return build(ErrorCode.RESOURCE_CONFLICT, ErrorCode.RESOURCE_CONFLICT.getDefaultMessage(),
                request.getRequestURI());
    }

    // ---------------------------------------------------------------- security
    // (when thrown from controllers)
    @ExceptionHandler(AuthenticationException.class)
    public ResponseEntity<ApiError> handleAuthentication(AuthenticationException ex, HttpServletRequest request) {
        return build(ErrorCode.UNAUTHORIZED, ErrorCode.UNAUTHORIZED.getDefaultMessage(), request.getRequestURI());
    }

    @ExceptionHandler(AccessDeniedException.class)
    public ResponseEntity<ApiError> handleAccessDenied(AccessDeniedException ex, HttpServletRequest request) {
        return build(ErrorCode.FORBIDDEN, ErrorCode.FORBIDDEN.getDefaultMessage(), request.getRequestURI());
    }

    // ---------------------------------------------------------------- catch-all
    @ExceptionHandler(Exception.class)
    public ResponseEntity<ApiError> handleUnexpected(Exception ex, HttpServletRequest request) {
        log.atError().addKeyValue("path", request.getRequestURI())
                .log("Unhandled exception", ex);
        return build(ErrorCode.INTERNAL_ERROR, ErrorCode.INTERNAL_ERROR.getDefaultMessage(),
                request.getRequestURI());
    }

    // ---------------------------------------------------------------- Spring MVC
    // exceptions
    @Override
    protected ResponseEntity<Object> handleMethodArgumentNotValid(MethodArgumentNotValidException ex,
            HttpHeaders headers,
            HttpStatusCode status,
            WebRequest request) {
        List<ApiError.FieldViolation> violations = ex.getBindingResult().getFieldErrors().stream()
                .map(fe -> new ApiError.FieldViolation(fe.getField(), fe.getDefaultMessage()))
                .toList();
        ApiError body = ApiError.of(status, ErrorCode.VALIDATION_ERROR.name(),
                ErrorCode.VALIDATION_ERROR.getDefaultMessage(), path(request), violations);
        log.atWarn().addKeyValue("path", path(request))
                .addKeyValue("violationCount", violations.size())
                .log("Request validation failed");
        return ResponseEntity.status(status).headers(headers).body(body);
    }

    /**
     * Fallback for every other Spring MVC exception: keep status chosen by Spring,
     * unify the body.
     */
    @Override
    protected ResponseEntity<Object> handleExceptionInternal(Exception ex, Object body, HttpHeaders headers,
            HttpStatusCode statusCode, WebRequest request) {
        ProblemDetail problem = body instanceof ProblemDetail pd ? pd
                : ex instanceof ErrorResponse er ? er.getBody() : null;

        String message;
        String code;
        if (statusCode.is5xxServerError()) {
            log.atError().addKeyValue("path", path(request)).log("Framework error", ex);
            message = ErrorCode.INTERNAL_ERROR.getDefaultMessage();
            code = ErrorCode.INTERNAL_ERROR.name();
        } else {
            message = problem != null && problem.getDetail() != null ? problem.getDetail() : "Bad request";
            HttpStatus resolved = HttpStatus.resolve(statusCode.value());
            code = resolved != null ? resolved.name() : "ERROR";
            log.atWarn().addKeyValue("path", path(request)).addKeyValue("errorCode", code)
                    .log("Request rejected: {}", ex.getClass().getSimpleName());
        }
        ApiError error = ApiError.of(statusCode, code, message, path(request));
        return ResponseEntity.status(statusCode).headers(headers).body(error);
    }

    // ---------------------------------------------------------------- helpers
    private ResponseEntity<ApiError> build(ErrorCode code, String message, String path) {

package tutorjava.digitalbanking.idempotency;

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;
import tutorjava.digitalbanking.common.exception.BusinessException;
import tutorjava.digitalbanking.common.exception.ErrorCode;
import tutorjava.digitalbanking.common.util.Sha256;
import tutorjava.digitalbanking.config.IdempotencyProperties;

import java.time.Instant;
import java.util.UUID;
import java.util.regex.Pattern;

/**
 * Exactly-once semantics for money-moving requests.
 *
 * The claim row and the business change live in ONE database transaction:
 * - success -> both commit together (key + result recorded atomically)
 * - failure -> both roll back (client may retry with the same key and it is
 * re-evaluated)
 * - duplicate (sequential or concurrent) -> the unique index makes the loser
 * wait, then replay the stored result
 * - same key with another payload -> rejected (422), protects against client
 * bugs
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class IdempotencyService {

    public static final String HEADER = "Idempotency-Key";

    private static final Pattern KEY_PATTERN = Pattern.compile("^[A-Za-z0-9_.:-]{8,100}$");

    private final IdempotencyKeyRepository repository;
    private final IdempotencyProperties properties;

    @Transactional(propagation = Propagation.MANDATORY)
    public IdempotencyClaim claim(UUID userId, String key, String requestFingerprint) {
        if (key == null || !KEY_PATTERN.matcher(key).matches()) {
            throw new BusinessException(ErrorCode.INVALID_IDEMPOTENCY_KEY);
        }
        String requestHash = Sha256.hex(requestFingerprint);
        Instant now = Instant.now();
        UUID claimId = UUID.randomUUID();

        int inserted = repository.insertIfAbsent(claimId, userId, key, requestHash, now,
                now.plus(properties.ttl()));
        if (inserted == 1) {
            return new IdempotencyClaim.Fresh(claimId);
        }

        IdempotencyKey existing = repository.findByUserIdAndIdempotencyKey(userId, key)
                .orElseThrow(() -> new BusinessException(ErrorCode.IDEMPOTENCY_IN_PROGRESS));
        if (!existing.getRequestHash().equals(requestHash)) {
            log.atWarn().addKeyValue("userId", userId).log("Idempotency-Key reused with a different payload");
            throw new BusinessException(ErrorCode.IDEMPOTENCY_KEY_REUSED);
        }
        if (existing.getTransactionId() == null) {
            throw new BusinessException(ErrorCode.IDEMPOTENCY_IN_PROGRESS);
        }
        log.atInfo().addKeyValue("userId", userId)
                .addKeyValue("transactionId", existing.getTransactionId())
                .log("Idempotent replay: returning stored result");
        return new IdempotencyClaim.Replay(existing.getTransactionId());
    }

    @Transactional(propagation = Propagation.MANDATORY)
    public void complete(UUID claimId, UUID transactionId) {
        repository.attachTransaction(claimId, transactionId);
    }

}

import org.springframework.core.annotation.Order;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;

/**
 * Runs before Spring Security. Responsibilities:
 * 1. correlation id: accept a (sanitised) X-Request-Id or generate one, echo it
 * back, put it in the MDC;
 * 2. one structured access-log line per request (method, path, status,
 * duration, user).
 * Every log line written while handling the request automatically carries
 * requestId/userId via the MDC.
 */
@Slf4j
@Component
@Order(Ordered.HIGHEST_PRECEDENCE)
public class RequestLoggingFilter extends OncePerRequestFilter {

    public static final String REQUEST_ID_HEADER = "X-Request-Id";
    public static final String MDC_REQUEST_ID = "requestId";
    public static final String MDC_USER_ID = "userId";

    /** Prevents log injection through a client supplied id. */
    private static final Pattern SAFE_REQUEST_ID = Pattern.compile("^[A-Za-z0-9._-]{8,64}$");

    @Override
    protected void doFilterInternal(HttpServletRequest request, HttpServletResponse response,
            FilterChain chain) throws ServletException, IOException {
        String requestId = resolveRequestId(request.getHeader(REQUEST_ID_HEADER));
        MDC.put(MDC_REQUEST_ID, requestId);
        response.setHeader(REQUEST_ID_HEADER, requestId);

        long startNanos = System.nanoTime();
        try {
            chain.doFilter(request, response);
        } finally {
            long durationMs = (System.nanoTime() - startNanos) / 1_000_000;
            String path = request.getRequestURI();
            var event = path.startsWith("/actuator") ? log.atDebug() : log.atInfo();
            event.addKeyValue("method", request.getMethod())
                    .addKeyValue("path", path)
                    .addKeyValue("status", response.getStatus())
                    .addKeyValue("durationMs", durationMs)
                    .log("HTTP request completed");
            MDC.clear();
        }
    }

    private String resolveRequestId(String incoming) {
        return incoming != null && SAFE_REQUEST_ID.matcher(incoming).matches()
                ? incoming
                : UUID.randomUUID().toString();
    }
}

import tutorjava.digitalbanking.config.JwtProperties;
import tutorjava.digitalbanking.user.User;
import tutorjava.digitalbanking.user.UserRole;

import java.util.Base64;
import java.util.Date;
import java.util.Optional;
import java.util.UUID;

/**
 * Issues and validates HS256 access tokens (Nimbus JOSE+JWT).
 * Validation pins the algorithm, verifies the signature, issuer and expiry.
 * Fails fast at startup when the secret is missing, not base64 or shorter than
 * 256 bits.
 */
@Slf4j
@Service
public class JwtService {

    private static final String CLAIM_EMAIL = "email";
    private static final String CLAIM_ROLE = "role";

    private final JwtProperties properties;
    private final JWSSigner signer;
    private final JWSVerifier verifier;

    public JwtService(JwtProperties properties) {
        this.properties = properties;
        try {
            byte[] secret = Base64.getDecoder().decode(properties.secret());
            if (secret.length < 32) {
                throw new IllegalStateException("app.jwt.secret must decode to at least 32 bytes (256 bits)");
            }
            this.signer = new MACSigner(secret);
            this.verifier = new MACVerifier(secret);
        } catch (IllegalArgumentException e) {
            throw new IllegalStateException("app.jwt.secret must be base64 encoded (openssl rand -base64 48)", e);
        } catch (JOSEException e) {
            throw new IllegalStateException("Unable to initialise JWT signer/verifier", e);
        }
    }

    public IssuedToken generateAccessToken(User user) {
        Instant now = Instant.now();
        Instant expiry = now.plus(properties.accessTokenExpiration());

        JWTClaimsSet claims = new JWTClaimsSet.Builder()
                .issuer(properties.issuer())
                .subject(user.getId().toString())
                .jwtID(UUID.randomUUID().toString())
                .issueTime(Date.from(now))
                .expirationTime(Date.from(expiry))
                .claim(CLAIM_EMAIL, user.getEmail())
                .claim(CLAIM_ROLE, user.getRole().name())
                .build();

        try {
            SignedJWT jwt = new SignedJWT(new JWSHeader(JWSAlgorithm.HS256), claims);
            jwt.sign(signer);
            return new IssuedToken(jwt.serialize(), properties.accessTokenExpiration().toSeconds());
        } catch (JOSEException e) {
            throw new IllegalStateException("Unable to sign JWT", e);
        }
    }

    /**
     * Returns the principal for a valid token, or empty for anything wrong (never
     * throws).
     */
    public Optional<AuthenticatedUser> parseAndValidate(String token) {
        try {
            SignedJWT jwt = SignedJWT.parse(token);
            if (!JWSAlgorithm.HS256.equals(jwt.getHeader().getAlgorithm()) || !jwt.verify(verifier)) {
                log.debug("JWT rejected: bad algorithm or signature");
                return Optional.empty();
            }
            JWTClaimsSet claims = jwt.getJWTClaimsSet();
            Date expiration = claims.getExpirationTime();
            if (!properties.issuer().equals(claims.getIssuer())
                    || expiration == null
                    || expiration.toInstant().isBefore(Instant.now())) {
                log.debug("JWT rejected: bad issuer or expired");
                return Optional.empty();
            }
            return Optional.of(new AuthenticatedUser(
                    UUID.fromString(claims.getSubject()),
                    claims.getStringClaim(CLAIM_EMAIL),
                    UserRole.valueOf(claims.getStringClaim(CLAIM_ROLE))));
        } catch (ParseException | JOSEException | IllegalArgumentException | NullPointerException e) {
            log.debug("JWT rejected: {}", e.getClass().getSimpleName());
            return Optional.empty();
        }
    }
}

/**
 * Internal, transport-independent description of "what the user wants to do".
 * The controller maps each request DTO to a command; processors only know about commands.
 */
public record TransactionCommand(
        UUID userId,
        TransactionType type,
        String sourceAccountNumber,
        String destinationAccountNumber,
        BigDecimal amount,
        String description) {

    public TransactionCommand {
        // request validation already limits the scale to 2, so this never rounds
        amount = amount.setScale(2, RoundingMode.UNNECESSARY);
        description = description == null || description.isBlank() ? null : description.trim();
    }

    public static TransactionCommand deposit(UUID userId, String accountNumber, BigDecimal amount, String description) {
        return new TransactionCommand(userId, TransactionType.DEPOSIT, null, accountNumber, amount, description);
    }

    public static TransactionCommand withdrawal(UUID userId, String accountNumber, BigDecimal amount, String description) {
        return new TransactionCommand(userId, TransactionType.WITHDRAWAL, accountNumber, null, amount, description);
    }

    public static TransactionCommand transfer(UUID userId, String source, String destination,
                                              BigDecimal amount, String description) {
        return new TransactionCommand(userId, TransactionType.TRANSFER, source, destination, amount, description);
    }

    /** Canonical string of everything that defines the request; hashed for idempotency comparison. */
    public String fingerprint() {
        return String.join("|", type.name(),
                String.valueOf(sourceAccountNumber),
                String.valueOf(destinationAccountNumber),
                amount.toPlainString(),
                String.valueOf(description));
    }

import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import tutorjava.digitalbanking.common.exception.BusinessException;
import tutorjava.digitalbanking.common.exception.ErrorCode;
import tutorjava.digitalbanking.config.BankingProperties;
import tutorjava.digitalbanking.idempotency.IdempotencyClaim;
import tutorjava.digitalbanking.idempotency.IdempotencyService;
import tutorjava.digitalbanking.transaction.dto.TransactionResponse;
import tutorjava.digitalbanking.transaction.processor.TransactionProcessorRegistry;

/**
 * FACADE / orchestrator for every money movement.
 *
 * ONE @Transactional boundary wraps: idempotency claim -> account row locks -> balance changes ->
 * transaction + ledger rows -> idempotency completion. Any failure rolls back ALL of it (atomicity),
 * balances are protected by row locks (isolation) and CHECK constraints (consistency).
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class TransactionService {

    private final IdempotencyService idempotencyService;
    private final TransactionProcessorRegistry processorRegistry;
    private final TransactionRepository transactionRepository;
    private final BankingProperties bankingProperties;

    @Transactional
    public TransactionResult execute(String idempotencyKey, TransactionCommand command) {
        if (command.amount().compareTo(bankingProperties.maxTransactionAmount()) > 0) {
            throw new BusinessException(ErrorCode.AMOUNT_LIMIT_EXCEEDED);
        }

        IdempotencyClaim claim = idempotencyService.claim(command.userId(), idempotencyKey, command.fingerprint());

        return switch (claim) {
            case IdempotencyClaim.Fresh fresh -> {
                Transaction transaction = processorRegistry.get(command.type()).process(command);
                // idempotency_keys.transaction_id is a FK: the transaction row must physically exist
                // (and balance updates are sent to the DB) before the bulk UPDATE below runs.
                transactionRepository.flush();
                idempotencyService.complete(fresh.claimId(), transaction.getId());
                log.atInfo()
                        .addKeyValue("referenceNo", transaction.getReferenceNo())
                        .addKeyValue("type", transaction.getType())
                        .addKeyValue("amount", transaction.getAmount())
                        .addKeyValue("userId", command.userId())
                        .log("Transaction completed");
                yield new TransactionResult(TransactionResponse.from(transaction), false);
            }
            case IdempotencyClaim.Replay replay -> {
                Transaction transaction = transactionRepository.findById(replay.transactionId())
                        .orElseThrow(() -> new BusinessException(ErrorCode.TRANSACTION_NOT_FOUND));
                yield new TransactionResult(TransactionResponse.from(transaction), true);
            }
        };
    }
}

/** FACTORY / registry: resolves the strategy for a type. Spring injects every TransactionProcessor bean. */
@Component
public class TransactionProcessorRegistry {

    private final Map<TransactionType, TransactionProcessor> processors = new EnumMap<>(TransactionType.class);

    public TransactionProcessorRegistry(List<TransactionProcessor> available) {
        for (TransactionProcessor processor : available) {
            TransactionProcessor previous = processors.put(processor.supportedType(), processor);
            if (previous != null) {
                throw new IllegalStateException("Duplicate processor for " + processor.supportedType());
            }
        }
        for (TransactionType type : TransactionType.values()) {
            if (!processors.containsKey(type)) {
                throw new IllegalStateException("No processor registered for " + type);
            }
        }
    }

    public TransactionProcessor get(TransactionType type) {
        return processors.get(type);
    }
}

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Component;
import tutorjava.digitalbanking.account.Account;
import tutorjava.digitalbanking.transaction.EntryDirection;
import tutorjava.digitalbanking.transaction.LedgerEntry;

import tutorjava.digitalbanking.transaction.Transaction;
import tutorjava.digitalbanking.transaction.TransactionCommand;
import tutorjava.digitalbanking.transaction.TransactionRepository;

import java.security.SecureRandom;
import java.time.LocalDate;
import java.time.ZoneOffset;
import java.time.format.DateTimeFormatter;
import java.util.ArrayList;
import java.util.List;

/**
 * Shared persistence step of every processor: writes the transaction header and one ledger entry per
 * affected account. Call it AFTER the account balances have been updated so balance_after is correct.
 */
@Component
@RequiredArgsConstructor
public class TransactionRecorder {

    private static final String ALPHABET = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
    private static final DateTimeFormatter DATE = DateTimeFormatter.BASIC_ISO_DATE;

    private final TransactionRepository transactionRepository;
    private final LedgerEntryRepository ledgerEntryRepository;
    private final SecureRandom random = new SecureRandom();

    /** @param source null for deposits; @param destination null for withdrawals */
    public Transaction record(TransactionCommand command, Account source, Account destination) {
        String currency = source != null ? source.getCurrency() : destination.getCurrency();

        Transaction transaction = transactionRepository.save(Transaction.builder()
                .referenceNo(nextReference())
                .type(command.type())
                .amount(command.amount())
                .currency(currency)
                .sourceAccountNumber(source != null ? source.getAccountNumber() : null)
                .destinationAccountNumber(destination != null ? destination.getAccountNumber() : null)
                .description(command.description())
                .createdBy(command.userId())
                .build());

        List<LedgerEntry> entries = new ArrayList<>(2);
        if (source != null) {
            entries.add(LedgerEntry.of(transaction, source.getId(), EntryDirection.DEBIT,
                    command.amount(), source.getBalance()));
        }
        if (destination != null) {
            entries.add(LedgerEntry.of(transaction, destination.getId(), EntryDirection.CREDIT,
                    command.amount(), destination.getBalance()));
        }
        ledgerEntryRepository.saveAll(entries);
        return transaction;
    }

    private String nextReference() {
        StringBuilder sb = new StringBuilder("TRX")
                .append(LocalDate.now(ZoneOffset.UTC).format(DATE));
        for (int i = 0; i < 10; i++) {
            sb.append(ALPHABET.charAt(random.nextInt(ALPHABET.length())));
        }
        return sb.toString();
    }
}

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Component;
import tutorjava.digitalbanking.account.Account;
import tutorjava.digitalbanking.account.AccountLockService;
import tutorjava.digitalbanking.common.exception.BusinessException;
import tutorjava.digitalbanking.common.exception.ErrorCode;
import tutorjava.digitalbanking.transaction.Transaction;
import tutorjava.digitalbanking.transaction.TransactionCommand;
import tutorjava.digitalbanking.transaction.TransactionType;

import java.util.Map;
import java.util.Set;

/** Moves money from one of the caller's accounts to any other account, atomically. */
@Component
@RequiredArgsConstructor
public class TransferProcessor implements TransactionProcessor {

    private final AccountLockService accountLockService;
    private final TransactionRecorder recorder;

    @Override
    public TransactionType supportedType() {
        return TransactionType.TRANSFER;
    }

    @Override
    public Transaction process(TransactionCommand command) {
        String sourceNumber = command.sourceAccountNumber();
        String destinationNumber = command.destinationAccountNumber();
        if (sourceNumber.equals(destinationNumber)) {
            throw new BusinessException(ErrorCode.SAME_ACCOUNT_TRANSFER);
        }

        // Both rows are locked in one statement, in a deterministic order -> no deadlocks.
        Map<String, Account> locked = accountLockService.lock(Set.of(sourceNumber, destinationNumber));
        Account source = locked.get(sourceNumber);
        Account destination = locked.get(destinationNumber);

        if (!source.isOwnedBy(command.userId())) {
            throw new BusinessException(ErrorCode.ACCOUNT_NOT_FOUND);
        }
        if (!source.getCurrency().equals(destination.getCurrency())) {
            throw new BusinessException(ErrorCode.CURRENCY_MISMATCH);
        }

        source.debit(command.amount());
        destination.credit(command.amount());
        return recorder.record(command, source, destination);
    }
}

CREATE TABLE users (
    id                    UUID         PRIMARY KEY,
    email                 VARCHAR(254) NOT NULL,
    full_name             VARCHAR(100) NOT NULL,
    password_hash         VARCHAR(100) NOT NULL,
    role                  VARCHAR(20)  NOT NULL,
    enabled               BOOLEAN      NOT NULL DEFAULT TRUE,
    failed_login_attempts INTEGER      NOT NULL DEFAULT 0,
    locked_until          TIMESTAMPTZ,
    created_at            TIMESTAMPTZ  NOT NULL,
    updated_at            TIMESTAMPTZ  NOT NULL,
    version               BIGINT       NOT NULL DEFAULT 0,
    CONSTRAINT uq_users_email           UNIQUE (email),
    CONSTRAINT ck_users_email_lowercase CHECK (email = lower(email)),
    CONSTRAINT ck_users_role            CHECK (role IN ('USER', 'ADMIN')),
    CONSTRAINT ck_users_failed_attempts CHECK (failed_login_attempts >= 0)
);

CREATE TABLE accounts (
    id             UUID           PRIMARY KEY,
    account_number VARCHAR(10)    NOT NULL,
    user_id        UUID           NOT NULL REFERENCES users (id),
    account_name   VARCHAR(50)    NOT NULL,
    currency       VARCHAR(3)     NOT NULL,
    balance        NUMERIC(19, 2) NOT NULL DEFAULT 0,
    status         VARCHAR(20)    NOT NULL,
    created_at     TIMESTAMPTZ    NOT NULL,
    updated_at     TIMESTAMPTZ    NOT NULL,
    version        BIGINT         NOT NULL DEFAULT 0,
    CONSTRAINT uq_accounts_number  UNIQUE (account_number),
    CONSTRAINT ck_accounts_balance CHECK (balance >= 0),
    CONSTRAINT ck_accounts_status  CHECK (status IN ('ACTIVE', 'FROZEN', 'CLOSED'))
);

-- "my accounts" listing
CREATE INDEX idx_accounts_user_created ON accounts (user_id, created_at DESC);

CREATE TABLE transactions (
    id                         UUID           PRIMARY KEY,
    reference_no               VARCHAR(30)    NOT NULL,
    type                       VARCHAR(20)    NOT NULL,
    amount                     NUMERIC(19, 2) NOT NULL,
    currency                   VARCHAR(3)     NOT NULL,
    source_account_number      VARCHAR(10)    REFERENCES accounts (account_number),
    destination_account_number VARCHAR(10)    REFERENCES accounts (account_number),
    description                VARCHAR(140),
    created_by                 UUID           NOT NULL REFERENCES users (id),
    created_at                 TIMESTAMPTZ    NOT NULL,
    CONSTRAINT uq_transactions_reference UNIQUE (reference_no),
    CONSTRAINT ck_transactions_amount    CHECK (amount > 0),
    CONSTRAINT ck_transactions_shape CHECK (
        (type = 'DEPOSIT'    AND source_account_number IS NULL     AND destination_account_number IS NOT NULL)
     OR (type = 'WITHDRAWAL' AND source_account_number IS NOT NULL AND destination_account_number IS NULL)
     OR (type = 'TRANSFER'   AND source_account_number IS NOT NULL AND destination_account_number IS NOT NULL
                             AND source_account_number <> destination_account_number)
    )
);

-- Double-entry style ledger: one row per account affected by a transaction.
-- Account statements are served from this table using a single index.
CREATE TABLE ledger_entries (
    id             UUID           PRIMARY KEY,
    transaction_id UUID           NOT NULL REFERENCES transactions (id),
    account_id     UUID           NOT NULL REFERENCES accounts (id),
    direction      VARCHAR(10)    NOT NULL,
    amount         NUMERIC(19, 2) NOT NULL,
    balance_after  NUMERIC(19, 2) NOT NULL,
    created_at     TIMESTAMPTZ    NOT NULL,
    CONSTRAINT ck_ledger_direction     CHECK (direction IN ('DEBIT', 'CREDIT')),
    CONSTRAINT ck_ledger_amount        CHECK (amount > 0),
    CONSTRAINT ck_ledger_balance_after CHECK (balance_after >= 0)
);

CREATE INDEX idx_ledger_account_created ON ledger_entries (account_id, created_at DESC, id DESC);
CREATE INDEX idx_ledger_transaction     ON ledger_entries (transaction_id);

CREATE TABLE idempotency_keys (
    id              UUID         PRIMARY KEY,
    user_id         UUID         NOT NULL REFERENCES users (id),
    idempotency_key VARCHAR(100) NOT NULL,
    request_hash    VARCHAR(64)  NOT NULL,
    transaction_id  UUID         REFERENCES transactions (id),
    created_at      TIMESTAMPTZ  NOT NULL,
    expires_at      TIMESTAMPTZ  NOT NULL,
    CONSTRAINT uq_idempotency_user_key UNIQUE (user_id, idempotency_key)
);

CREATE INDEX idx_idempotency_expires_at ON idempotency_keys (expires_at);

-- Financial records are append-only: block UPDATE / DELETE at the database level.
CREATE FUNCTION forbid_mutation() RETURNS trigger AS $$
BEGIN
    RAISE EXCEPTION '% on % is not allowed (append-only table)', TG_OP, TG_TABLE_NAME;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_transactions_append_only
    BEFORE UPDATE OR DELETE ON transactions
    FOR EACH ROW EXECUTE FUNCTION forbid_mutation();

CREATE TRIGGER trg_ledger_entries_append_only
    BEFORE UPDATE OR DELETE ON ledger_entries
    FOR EACH ROW EXECUTE FUNCTION forbid_mutation();

-- Refresh tokens are opaque random strings. Only their SHA-256 hash is stored, so a database
-- leak does not leak usable tokens. Tokens belong to a "family" (= one login session): every refresh
-- rotates the token inside the same family, and replaying an already-used token revokes the family.
CREATE TABLE refresh_tokens (
    id                 UUID        PRIMARY KEY,
    user_id            UUID        NOT NULL REFERENCES users (id),
    family_id          UUID        NOT NULL,
    token_hash         VARCHAR(64) NOT NULL,
    session_started_at TIMESTAMPTZ NOT NULL,
    expires_at         TIMESTAMPTZ NOT NULL,
    revoked_at         TIMESTAMPTZ,
    created_at         TIMESTAMPTZ NOT NULL,
    CONSTRAINT uq_refresh_tokens_hash UNIQUE (token_hash)
);

CREATE INDEX idx_refresh_tokens_family  ON refresh_tokens (family_id);
CREATE INDEX idx_refresh_tokens_user    ON refresh_tokens (user_id);
CREATE INDEX idx_refresh_tokens_expires ON refresh_tokens (expires_at);
